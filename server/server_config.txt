# Gen1Online+ server settings: where it listens, and the game modes.
# Restart the server after editing.

# Where the server listens.  host = 0.0.0.0 takes every network (LAN,
# Tailscale, port forwarding); 127.0.0.1 only this PC.  Players type this
# PC's address (and the port, when it isn't 7779) in their game: START >
# CONNECT > SERVER ADDRESS.  --host and --port on the command line win.
host = 0.0.0.0
port = 7779

#
# The modes play on a Gen 1 world (Red, Blue and Yellow) and on a FireRed /
# LeafGreen world; a Crystal server ignores them.  Everyone on the server
# plays the same run with the same rules and, with the randomizer, the same
# shuffled world.  Turning a mode on
# (or a new run starting) restarts every player's ONLINE save from their
# bedroom; their offline saves are never touched, and the old online save is
# kept as a backup.

# Hardcore Nuzlocke:
#   - only the first wild Pokemon you meet in each area can be caught,
#     whatever it is (one per player; a species you already own counts too).
#     Encounters before you first have Poke Balls (Route 1) don't count; from
#     then on every area's first one does, even with no ball in your bag
#   - every player needs the current version of the mod: an older one is
#     told to update
#   - a Pokemon that faints is gone for good
#   - no items in battle (Poke Balls still work), battle style is SET,
#     and Pokemon stop gaining EXP at the level cap of the next gym (with
#     shuffled badges: the weakest gym leader you can reach and haven't beaten)
#   - if ANY player's whole party faints, the run is over for the whole team:
#     the server starts a new run (with a new seed) and everyone starts over
# nuzlocke = off | hardcore
nuzlocke = off

# Trades in the hardcore Nuzlocke: how many Pokemon each player may receive
# through the GTS, Wonder Trade or a link trade between two gym leaders.
# With 1, you may trade once; beating a gym leader lets you trade once more,
# until the next leader (your own gym leaders beaten, not the team's).  A
# GTS or Wonder Trade deposit is refused while your trades are used up; a
# Pokemon that reaches your claim box then waits for your next gym leader.
# nuzlocke_trades = off (no limit) | 1 | 2 | ...
nuzlocke_trades = off

# Co-op randomizer: wild Pokemon, items and gym badges are shuffled.
# A gym leader might hand over a POTION while the BOULDERBADGE lies on the
# floor somewhere or comes from an NPC.  Placement follows the game's
# progression, so a run can always be finished (nothing you need is left
# aboard the S.S. Anne, and the early routes always have wild Pokemon that
# can learn CUT).
# On FireRed and LeafGreen every wild Pokemon may be any of the 386 (Gen 1
# to 3; they evolve before the National Pokedex), and the item balls, hidden
# items, a few gifts (the TEA, the LIFT KEY, the SILPH SCOPE, HM06...) and
# the gym badges are shuffled.
# randomizer = on | off
randomizer = off

# What the randomizer shuffles (only when randomizer = on).  The starters
# become basic Pokemon that evolve twice, like the real ones (on Yellow too:
# Oak hands over something else than PIKACHU, which then doesn't follow you);
# the rival keeps his usual team.
randomize_encounters = on
randomize_items = on
randomize_badges = on
randomize_starters = on

# Legendary Pokemon in the wild (randomizer only): each wild encounter in
# grass, water, caves, fishing or Rock Smash has this chance of being a
# legendary instead, at that encounter's level (FireRed/LeafGreen: any of
# the 21, Lugia to Deoxys; Gen 1: the birds, MEWTWO and MEW).  Scripted
# battles (SNORLAX, the legendaries' own spots) stay as they are.
# wild_legendaries = off | on (1 in 100) | a percent, like 0.5 or 2
wild_legendaries = off

# Trainers' teams (randomizer only).  Each Pokemon becomes one of similar
# strength at the same level, never a legendary; every player meets the same
# teams.
#   gyms: gym leaders and the trainers in their gyms get the gym's type
#         (Brock Rock, Misty Water...), the Elite Four theirs (Ice, Fighting,
#         Ghost, Dragon), the Champion any type
#   on:   the same, and every other trainer gets random Pokemon too
# randomize_trainers = off | gyms | on
randomize_trainers = off

# The team shares key items, HMs and badges: whatever one player finds, every
# player gets.  auto = on whenever the randomizer is.
# shared_key_items = auto | on | off
shared_key_items = auto

# Multiworld (needs randomizer = on): every player gets their OWN world, and
# each badge, HM and key item exists in exactly ONE of them.  The BOULDERBADGE
# might lie in one friend's Mt. Moon and nowhere else; when that friend picks
# it up, the whole team has it.  The shuffle still guarantees the team can
# finish.  Wild Pokemon differ per world too.
#   - players = how many worlds; the first players to join each take one
#     and keep it (also across Nuzlocke runs).  Anyone else is turned away.
#   - everyone must play the same game's data (Red/Blue, or Yellow, or
#     FireRed/LeafGreen); the first player to join decides.
#   - a world's items are only reachable by its player: if someone stops
#     playing, the team may get stuck.
# multiworld = off | on
multiworld = off
players = 2
